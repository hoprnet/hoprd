7pub const DEFAULT_API_HOST: &str = "127.0.0.1";

11fn validate_api_auth(token: &Auth) -> Result<(), ValidationError> {
12    match &token {
13        Auth::None => Ok(()),
14        Auth::Token(token) => {
15            if token.len() >= MINIMAL_API_TOKEN_LENGTH {
16                Ok(())
17            } else {
18                Err(ValidationError::new("The API token is too short"))
19            }
20        }
21    }
22}

31#[derive(
32    Debug, Clone, PartialEq, smart_default::SmartDefault, Validate, Serialize, Deserialize,
33)]
34#[serde(deny_unknown_fields)]
35pub struct Api {
36    /// Selects whether the REST API is enabled
37    #[serde(default)]
38    pub enable: bool,
39    /// Auth enum holding the API auth configuration
40    #[validate(custom(function = "validate_api_auth"))]
41    #[serde(default)]
42    pub auth: Auth,
43    /// Host and port combination where the REST API should be located
44    #[validate(nested)]
45    #[serde(default = "default_api_host")]
46    #[default(default_api_host())]
47    pub host: HostConfig,
48    /// Enables the deprecated explicit-path session creation endpoint.
49    #[serde(default)]
50    pub enable_explicit_path_sessions: bool,
51    /// Flow-control (AIMD send-window) profile applied to sessions this node *initiates*
52    /// as a client via the session API (`off` | `clean` | `robust`).
53    ///
54    /// Flow control is an entry/sending-side mechanism: relays and exit nodes never run
55    /// it, so this setting has no effect on nodes used purely for relaying/exiting.
56    /// Default is `robust` — the tail-tolerance profile validated for throttled /
57    /// high-latency (multi-hop) paths.
58    #[serde(default)]
59    pub session_flow_control: SessionFlowControl,
60}
61
62/// Flow-control profile for node-initiated sessions. See the `api.session_flow_control` config option.
63#[derive(
64    Debug,
65    Clone,
66    Copy,
67    PartialEq,
68    Eq,
69    smart_default::SmartDefault,
70    Serialize,
71    Deserialize,
72    utoipa::ToSchema,
73)]
74#[serde(rename_all = "lowercase")]
75pub enum SessionFlowControl {
76    /// Flow control disabled — sends are hand-paced by the caller.
77    Off,
78    /// The verified clean profile (`FlowControlConfig::default`).
79    Clean,
80    /// The tail-tolerance bundle (persist probe + raised frame-retransmission budget) for
81    /// throttled / high-latency multi-hop paths (`FlowControlConfig::robust`).
82    #[default]
83    Robust,
84}
85
86impl SessionFlowControl {
87    /// Maps the profile to the protocol-level flow-control configuration.
88    pub fn to_config(self) -> Option<hopr_lib::exports::transport::FlowControlConfig> {
89        use hopr_lib::exports::transport::FlowControlConfig;
90        match self {
91            SessionFlowControl::Off => None,
92            SessionFlowControl::Clean => Some(FlowControlConfig::default()),
93            SessionFlowControl::Robust => Some(FlowControlConfig::robust()),
94        }
95    }
96}

98#[inline]
99fn default_api_host() -> HostConfig {
100    HostConfig::from_str(format!("{DEFAULT_API_HOST}:{DEFAULT_API_PORT}").as_str())
101        .expect("default credentials should always work")
102}
