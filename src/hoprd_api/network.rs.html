143/// Lists peers with immediate observation data from the network graph.
144///
145/// Returns only peers that have at least one edge with immediate QoS data,
146/// representing nodes the current node has direct transport observations for.
147#[utoipa::path(
148    get,
149    path = const_format::formatcp!("{BASE_PATH}/network/connected"),
150    description = "List connected peers with immediate observation data from the network graph",
151    responses(
152        (status = 200, description = "Connected peers with immediate observations", body = Vec<ConnectedPeerResponse>),
153        (status = 401, description = "Invalid authorization token.", body = ApiError),
154        (status = 422, description = "Unknown failure", body = ApiError)
155    ),
156    security(
157        ("api_token" = []),
158        ("bearer_token" = [])
159    ),
160    tag = "Network"
161)]
162pub(super) async fn connected<
163    H: HasChainApi<ChainError = hopr_lib::errors::HoprLibError> + HasGraphView + Send + Sync + 'static,
164>(
165    State(state): State<Arc<InternalState<H>>>,
166) -> impl IntoResponse {
167    let hopr = state.hopr.clone();
168    let graph = hopr.graph();
169    let edges = graph.connected_edges();
170
171    let me_key = graph.identity();
172
173    // Collect peers that are connected (is_connected == true) with immediate QoS data.
174    let mut peers = Vec::new();
175    for (src, dst, obs) in &edges {
176        if src != me_key {
177            continue;
178        }
179        let Some(imm) = obs.immediate_qos() else {
180            continue;
181        };
182        // Only edges observed to be up. An unchecked edge is not evidence of connectivity, so it is
183        // skipped here exactly as an observed-down one is.
184        if imm.is_connected() != Some(true) {
185            continue;
186        }
187
188        let address = match hopr.chain_api().packet_key_to_chain_key(dst) {
189            Ok(Some(addr)) => addr,
190            _ => continue,
191        };
192
193        peers.push(ConnectedPeerResponse {
194            address,
195            probe_rate: imm.average_probe_rate(),
196            last_update: obs.last_update().as_millis(),
197            average_latency: imm.average_latency().map(|l| l.as_millis()),
198            score: obs.score(),
199        });
200    }
201
202    (StatusCode::OK, Json(peers)).into_response()
203}

317/// Returns the network graph in DOT (Graphviz) format.
318///
319/// Only connected nodes (those with at least one edge) are included.
320/// Nodes are labeled by their on-chain (Ethereum) address when resolvable,
321/// falling back to the offchain public key hex representation.
322/// Edges carry quality annotations: score, latency (ms), and capacity when available.
323///
324/// Pass `?reachableOnly=true` to limit the output to edges reachable from this node.
325#[utoipa::path(
326    get,
327    path = const_format::formatcp!("{BASE_PATH}/network/graph"),
328    description = "Get the network graph in DOT (Graphviz) format",
329    params(GraphQueryRequest),
330    responses(
331        (status = 200, description = "DOT representation of the network graph", body = String, content_type = "text/plain"),
332        (status = 401, description = "Invalid authorization token.", body = ApiError),
333    ),
334    security(
335        ("api_token" = []),
336        ("bearer_token" = [])
337    ),
338    tag = "Network"
339)]
340pub(super) async fn graph<
341    H: HasChainApi<ChainError = hopr_lib::errors::HoprLibError> + HasGraphView + Send + Sync + 'static,
342>(
343    State(state): State<Arc<InternalState<H>>>,
344    Query(query): Query<GraphQueryRequest>,
345) -> impl IntoResponse {
346    let hopr = &state.hopr;
347    let graph = hopr.graph();
348
349    let edges = if query.reachable_only {
350        graph.reachable_edges()
351    } else {
352        graph.connected_edges()
353    };
354
355    // Build offchain key → onchain address mapping for all nodes in the graph.
356    let mut unique_keys = std::collections::HashSet::new();
357    for (src, dst, _) in &edges {
358        unique_keys.insert(*src);
359        unique_keys.insert(*dst);
360    }
361
362    let mut key_to_addr: HashMap<hopr_lib::api::types::crypto::types::OffchainPublicKey, String> =
363        HashMap::new();
364    for key in &unique_keys {
365        let label = match hopr.chain_api().packet_key_to_chain_key(key) {
366            Ok(Some(addr)) => addr.to_string(),
367            _ => key.to_string(),
368        };
369        key_to_addr.insert(*key, label);
370    }
371
372    let label = |key: &hopr_lib::api::types::crypto::types::OffchainPublicKey| {
373        key_to_addr
374            .get(key)
375            .cloned()
376            .unwrap_or_else(|| key.to_string())
377    };
378
379    // Render DOT (Graphviz) format inline using trait methods on the observations.
380    let mut dot = String::from("digraph hopr {\n");
381    for (src, dst, obs) in &edges {
382        let src_label = label(src);
383        let dst_label = label(dst);
384        // An unmeasured edge carries no score; omitting the attribute keeps it distinguishable from
385        // one that scored zero.
386        let mut attrs = Vec::new();
387        if let Some(score) = obs.score() {
388            attrs.push(format!("score={score:.2}"));
389        }
390        if let Some(imm) = obs.immediate_qos()
391            && let Some(latency) = imm.average_latency()
392        {
393            attrs.push(format!("lat={}ms", latency.as_millis()));
394        }
395        if let Some(inter) = obs.intermediate_qos()
396            && let Some(balance) = inter.balance()
397        {
398            attrs.push(format!("balance={balance}"));
399        }
400        use std::fmt::Write;
401        let _ = writeln!(
402            dot,
403            "  \"{src_label}\" -> \"{dst_label}\" [label=\"{}\"];",
404            attrs.join(" ")
405        );
406    }
407    dot.push_str("}\n");
408
409    (
410        StatusCode::OK,
411        [(axum::http::header::CONTENT_TYPE, "text/plain")],
412        dot,
413    )
414        .into_response()
415}
