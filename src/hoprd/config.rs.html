63// Ensures the node has an identity source to load or create keys from: either a
64// non-empty identity file path or a private key. Without this, an empty file path
65// (the default) only surfaces as an opaque filesystem error at keypair creation.
66fn validate_identity_source(identity: &Identity) -> Result<(), ValidationError> {
67    let has_file = !identity.file.trim().is_empty();
68    let has_key = identity
69        .private_key
70        .as_ref()
71        .is_some_and(|k| !k.trim().is_empty());
72
73    if has_file || has_key {
74        return Ok(());
75    }
76
77    let mut error = ValidationError::new("identity_source_missing");
78    error.message = Some(
79        "no identity source configured: provide an identity file via --identity \
80         (HOPRD_IDENTITY) or a private key via --privateKey (HOPRD_PRIVATE_KEY)"
81            .into(),
82    );
83    Err(error)
84}

174        delay_range: std::env::var("HOPR_INTERNAL_MIXER_DELAY_RANGE_IN_MS")
175            .ok()
176            .and_then(|v| v.trim().parse::<u64>().ok())
177            .map(Duration::from_millis)
178            .unwrap_or(defaults.delay_range),
179        capacity: std::env::var("HOPR_INTERNAL_MIXER_CAPACITY")
180            .ok()
181            .and_then(|v| v.trim().parse::<usize>().ok())
182            .filter(|&c| c > 0)
183            .unwrap_or(defaults.capacity),
184        ..defaults
185    }

188/// User-facing PIX global configuration (wraps upstream [`PixGlobalConfig`] which has `serde(skip)`).
189#[derive(Debug, Clone, PartialEq, smart_default::SmartDefault, Serialize, Deserialize)]
190#[serde(deny_unknown_fields)]
191pub struct UserPixGlobalConfig {
192    /// Number of parts an SSA is split into.
193    ///
194    /// Default is 4096.
195    #[default(4096)]
196    #[serde(default = "default_pix_num_ssa_parts")]
197    pub num_ssa_parts: usize,
198    /// Number of shares required to reconstruct an SSA part.
199    ///
200    /// Default is 128.
201    #[default(128)]
202    #[serde(default = "default_pix_ssa_part_size")]
203    pub ssa_part_size: usize,
204    /// Number of additional shares sent beyond `ssa_part_size`, to absorb packet loss.
205    ///
206    /// **Leave unset unless you have measured your return-path loss.** Unset derives the
207    /// surplus from `ssa_part_size`, sized to absorb 20 % of a polynomial's shares going
208    /// missing. An absolute count means a different loss tolerance at every `ssa_part_size`,
209    /// which is why upstream stopped defaulting it to one — so this passes `None` through
210    /// rather than resolving it here, where `ssa_part_size` might not be the one in effect.
211    ///
212    /// The surplus is billed: it travels in the negotiated `PixParams`, so the per-SSA quota
213    /// counts it and the deposit pays for it whether or not any share is lost. Raising it buys
214    /// loss tolerance and costs money.
215    ///
216    /// Default is unset.
217    #[serde(default = "default_pix_additional_shares")]
218    pub additional_shares: Option<usize>,
219    /// Maximum number of SSA commitments this node, acting as an Entry, accepts in one request
220    /// from an Exit. Each accepted entry costs its own commitment, packet burst and on-chain
221    /// deposit, so this caps how much work one inbound packet can amplify into.
222    ///
223    /// Must be at least the `incoming_session_pix.ssas_per_request` of every Exit this node
224    /// uses: the batch size is not negotiated, so an Exit batching above this has every request
225    /// refused and loses the Session. Raise the two together.
226    ///
227    /// Clamped upstream to 1..=`MAX_SSA_BATCH_SIZE`, which hoprnet lowered from 20 to 9: each
228    /// entry in a batch costs the Entry a full commitment and its own on-chain deposit, and the
229    /// Exit a live reconstructor cycle of ~49 MiB.
230    ///
231    /// Default is 2.
232    #[default(default_pix_max_ssas_per_request())]
233    #[serde(default = "default_pix_max_ssas_per_request")]
234    pub max_ssas_per_request: usize,
235}

250/// User-facing incoming session PIX configuration (a flattened, curated [`IncomingSessionPixConfig`]).
251///
252/// Two things are flattened. The quota window is a `RangeInclusive` upstream, which no YAML file
253/// can state, so it is split into its two bounds. And the deadlines and batch settings moved one
254/// level down into a [`SupervisorConfig`] — upstream groups them because they are validated against
255/// each other and against the Exit's reconstructor as a set — which this block hoists back up, so
256/// an existing `incoming_session_pix` stanza keeps parsing unchanged.
257///
258/// The subset is deliberate rather than an omission. hoprd pins the Exit-side reconstructor to
259/// upstream's defaults (see `reconstructor` in the conversion below), so the supervision dials that
260/// only mean something *relative to it* — `max_recovery_idle`, `tombstone_retention_window` — are
261/// left at defaults chosen to pair with those. What is exposed here is what an operator can move
262/// without also having to move something hoprd does not surface.
263#[derive(Debug, Clone, PartialEq, smart_default::SmartDefault, Serialize, Deserialize)]
264#[serde(deny_unknown_fields)]
265pub struct UserIncomingSessionPixConfig {
266    /// Reject incoming Sessions that do not opt into PIX.
267    #[default(false)]
268    #[serde(default)]
269    pub enforce_pix: bool,
270    /// Minimum acceptable PIX data quota in bytes.
271    ///
272    /// What this is compared against depends on
273    /// [`allow_dynamic_ssa_batches`](Self::allow_dynamic_ssa_batches): with it on, the Exit looks
274    /// for a batch size whose *total* `batch × quota_per_ssa` lands in this window; with it off,
275    /// the Entry's per-SSA quota itself has to.
276    ///
277    /// Default is upstream's, ≈162 MiB.
278    #[default(default_pix_quota_range_min())]
279    #[serde(default = "default_pix_quota_range_min")]
280    pub quota_range_min: u64,
281    /// Maximum acceptable PIX data quota in bytes.
282    ///
283    /// Widening this is not free, and not only in traffic: upstream refuses a configuration whose
284    /// [`max_recovery_time`](Self::max_recovery_time) cannot cover one cycle at the top of this
285    /// window, or whose [`max_live_cycle_bytes`](Self::max_live_cycle_bytes) could not hold a
286    /// single Session offering it. Both are exposed below precisely so that raising this one stays
287    /// possible.
288    ///
289    /// Default is upstream's, ≈649 MiB.
290    #[default(default_pix_quota_range_max())]
291    #[serde(default = "default_pix_quota_range_max")]
292    pub quota_range_max: u64,
293    /// Maximum time to wait for SSA commitment delivery.
294    ///
295    /// Default is 20 seconds.
296    #[default(default_pix_max_ssa_delivery_time())]
297    #[serde(
298        default = "default_pix_max_ssa_delivery_time",
299        with = "humantime_serde"
300    )]
301    pub max_ssa_delivery_time: Duration,
302    /// Maximum time to wait for deposit into SSA.
303    ///
304    /// Default is 60 seconds.
305    #[default(default_pix_max_deposit_wait())]
306    #[serde(default = "default_pix_max_deposit_wait", with = "humantime_serde")]
307    pub max_deposit_wait: Duration,
308    /// Ceiling on the live Exit-side reconstructor state this node commits to, in bytes.
309    ///
310    /// This, and not `maximum_sessions`, is what bounds how many PIX Sessions the node accepts: a
311    /// Session reserves against this budget when it is admitted, at the dimensions the *peer*
312    /// offered, and one that does not fit is refused before any state is allocated for it.
313    ///
314    /// Default is upstream's 3 GiB, which admits ≈37 concurrent Sessions at the default PIX
315    /// dimensions.
316    #[default(default_pix_max_live_cycle_bytes())]
317    #[serde(default = "default_pix_max_live_cycle_bytes")]
318    pub max_live_cycle_bytes: u64,
319    /// Absolute per-SSA recovery deadline: a resource backstop on the Session slot and the
320    /// reconstructor state a cycle holds, not the anti-drip mechanism.
321    ///
322    /// It has to cover a whole cycle at the widest quota this node accepts, so it is tied to
323    /// [`quota_range_max`](Self::quota_range_max) and validated against it at load.
324    ///
325    /// Default is 2 hours.
326    #[default(default_pix_max_recovery_time())]
327    #[serde(default = "default_pix_max_recovery_time", with = "humantime_serde")]
328    pub max_recovery_time: Duration,
329    /// Packets the Exit will serve without a single PIX share coming back before its egress gate
330    /// blocks, as a backstop against an Entry that consumes service and returns nothing.
331    ///
332    /// **This has to clear the Session's SURB buffer whenever that buffer is deeper than one
333    /// cycle's paid surplus, and nothing validates that it does.** A PIX share is bound to a SURB
334    /// when the SURB is minted, and the Exit spends its buffer roughly in order, so the buffer is a
335    /// pipeline delay between a share being generated and being delivered. When a cycle recovers,
336    /// the SURBs already queued still carry *its* shares, and the successor's first share is that
337    /// whole queue away.
338    ///
339    /// Upstream credits part of that drain as liveness — the recovered cycle is kept as the paid
340    /// front until its FIFO tail passes — but only up to `num_ssa_parts × additional_shares`, the
341    /// surplus the cycle was actually paid for. That cap is a security bound, not a shortfall:
342    /// reconstruction releases the share set, so duplicate detection is gone and an uncapped
343    /// allowance would let an Entry replay one completed polynomial for unbounded service.
344    ///
345    /// So a Session whose SURB buffer exceeds that surplus volume has a remainder no protocol rule
346    /// can credit, and this is the only thing standing between it and a deadlock: the gate blocks
347    /// partway through the drain, and a blocked Exit spends no SURBs, which is the only thing that
348    /// was draining it. The Session then sits until upstream's 60-second `max_recovery_idle` closes
349    /// it, having recovered exactly one cycle.
350    ///
351    /// The default suits a Session buffering a few thousand SURBs. One provisioning a deep buffer
352    /// for throughput has to raise this to match it — see `session_pix_soak.rs`, which derives both
353    /// from the same rate.
354    ///
355    /// Default is upstream's 2048 packets.
356    #[default(default_pix_max_served_without_progress())]
357    #[serde(default = "default_pix_max_served_without_progress")]
358    pub max_served_without_progress: u64,
359    /// Number of SSAs this node, acting as an Exit, asks the Entry to commit to in one request.
360    ///
361    /// Batching amortizes the round trip over several deposit cycles, at the cost of holding that
362    /// many reconstructor cycles at once and serving that many SSA quotas unincentivized before
363    /// the first deposit lands. The kill switch scales with it: each cycle in a batch is given
364    /// `ssas_per_request × (max_deposit_wait + max_ssa_delivery_time)`.
365    ///
366    /// With [`allow_dynamic_ssa_batches`](Self::allow_dynamic_ssa_batches) on — the default — this
367    /// is a *ceiling* rather than the size asked for; see there.
368    ///
369    /// Must not exceed the peer Entry's `pix.max_ssas_per_request` — see that field.
370    ///
371    /// Clamped upstream to 1..=`MAX_SSA_BATCH_SIZE`, which hoprnet lowered from 20 to 9: each
372    /// entry in a batch costs the Entry a full commitment and its own on-chain deposit, and the
373    /// Exit a live reconstructor cycle of ~41 MiB.
374    ///
375    /// Default is 1, which reproduces the unbatched exchange exactly.
376    #[default(default_pix_ssas_per_request())]
377    #[serde(default = "default_pix_ssas_per_request")]
378    pub ssas_per_request: usize,
379    /// Let Session admission derive the batch size from the quota the Entry offered, instead of
380    /// always asking for [`ssas_per_request`](Self::ssas_per_request) of them.
381    ///
382    /// On, the Exit tries batch sizes from one upwards and takes the first whose total quota fits
383    /// the window above — the smallest fit, so it asks for no more Entry work, deposits or Exit
384    /// memory than the offer needs, and is less likely to cross the Entry's unadvertised cap.
385    ///
386    /// Off, `ssas_per_request` is the exact batch size and the per-SSA quota alone has to fit the
387    /// window. That is the behaviour before upstream added this, and it is what a deployment
388    /// wanting a *known* batch size — a test asserting on one, say — has to select.
389    ///
390    /// Default is upstream's, on.
391    #[default(default_pix_allow_dynamic_ssa_batches())]
392    #[serde(default = "default_pix_allow_dynamic_ssa_batches")]
393    pub allow_dynamic_ssa_batches: bool,
394}

527impl From<UserHoprLibConfig> for HoprLibConfig {
528    fn from(value: UserHoprLibConfig) -> Self {
529        let supervision_defaults = SupervisorConfig::default();
530        HoprLibConfig {
531            host: value.host,
532            publish: value.announce,
533            safe_module: value.safe_module,
534            protocol: HoprProtocolConfig {
535                transport: TransportConfig {
536                    announce_local_addresses: value.network.announce_local_addresses,
537                    prefer_local_addresses: value.network.prefer_local_addresses,
538                },
539                packet: HoprPacketPipelineConfig {
540                    codec: HoprCodecConfig {
541                        outgoing_win_prob: value
542                            .network
543                            .outgoing_ticket_winning_prob
544                            .and_then(|v| WinningProbability::try_from_f64(v).ok()),
545                        min_incoming_ticket_price: value.network.min_incoming_ticket_price,
546                        ..Default::default()
547                    },
548                    // Reply with the freshest SURBs first, so a return-path change takes effect
549                    // immediately instead of only after a stale backlog has been drained.
550                    //
551                    // Not under PIX, which needs the opposite end. A PIX share is delivered to the
552                    // reconstructor only when its SURB is *used*, and the ring buffer evicts from
553                    // the oldest end — so popping newest-first leaves the oldest SURBs unspent
554                    // until they are overwritten, and each overwrite is a permanently lost share.
555                    // Under sustained traffic the Exit then never assembles `ssa_part_size` shares
556                    // for any polynomial: the key is never reconstructed, nothing is swept, and the
557                    // Entry never gets a second deposit request. Observed exactly that — one
558                    // deposit confirmed, then a stall with no error anywhere — before this gate.
559                    //
560                    // `Fifo` is upstream's default for this reason; see `SurbPopOrder`, whose own
561                    // documentation calls out the lost-share hazard.
562                    //
563                    // Gated on the feature rather than on `enforce_pix` so the pop order is a
564                    // property of the binary rather than something a config toggle changes
565                    // underneath a running deployment. The cost is that a PIX-capable build whose
566                    // strategy list has no `Pix` stanza also gets `Fifo` — that is upstream's
567                    // default, so it forgoes an optimisation rather than regressing.
568                    surb_store: SurbStoreConfig {
569                        #[cfg(not(feature = "pix"))]
570                        pop_order: SurbPopOrder::Lifo,
571                        #[cfg(feature = "pix")]
572                        pop_order: SurbPopOrder::Fifo,
573                        ..Default::default()
574                    },
575                    ..Default::default()
576                },
577                // Simulated transit-latency shim (testing only); production leaves it disabled.
578                transit_latency: None,
579                probe: ProbeConfig {
580                    interval: value.network.probe_interval,
581                    recheck_threshold: value.network.probe_recheck_threshold,
582                    ..Default::default()
583                },
584                session: SessionGlobalConfig {
585                    idle_timeout: value.network.session_idle_timeout,
586                    establish_max_retries: value.network.session_establish_max_retries,
587                    tag_allocator: TagAllocatorConfig {
588                        session: value.network.maximum_sessions as u64,
589                        ..Default::default()
590                    },
591                    ..Default::default()
592                },
593                pix: PixGlobalConfig {
594                    num_ssa_parts: value.network.pix.num_ssa_parts,
595                    ssa_part_size: value.network.pix.ssa_part_size,
596                    additional_shares: value.network.pix.additional_shares,
597                    max_ssas_per_request: value.network.pix.max_ssas_per_request,
598                    // Exit-side reconstructor capacity is not yet exposed in `UserPixGlobalConfig`,
599                    // so operators get upstream's defaults. Listed rather than swept up by
600                    // `..Default::default()` on purpose: the exhaustive literal is what turns a new
601                    // upstream field into a compile error here instead of a silent default.
602                    reconstructor: Default::default(),
603                },
604                incoming_session_pix_config: IncomingSessionPixConfig {
605                    enforce_pix: value.network.incoming_session_pix.enforce_pix,
606                    quota_range: value.network.incoming_session_pix.quota_range_min
607                        ..=value.network.incoming_session_pix.quota_range_max,
608                    max_live_cycle_bytes: value.network.incoming_session_pix.max_live_cycle_bytes,
609                    supervision: SupervisorConfig {
610                        max_ssa_delivery_time: value
611                            .network
612                            .incoming_session_pix
613                            .max_ssa_delivery_time,
614                        max_deposit_wait: value.network.incoming_session_pix.max_deposit_wait,
615                        max_recovery_time: value.network.incoming_session_pix.max_recovery_time,
616                        ssas_per_request: value.network.incoming_session_pix.ssas_per_request,
617                        allow_dynamic_ssa_batches: value
618                            .network
619                            .incoming_session_pix
620                            .allow_dynamic_ssa_batches,
621                        max_served_without_progress: value
622                            .network
623                            .incoming_session_pix
624                            .max_served_without_progress,
625                        // The rest of the supervision policy is left where upstream put it, and
626                        // named rather than swept up by `..Default::default()` for the reason
627                        // `reconstructor` above is: the exhaustive literal is what turns a new
628                        // upstream field into a compile error here instead of a silent default.
629                        //
630                        // Two of them are the reconstructor's other half — `max_recovery_idle` must
631                        // sit between its `max_ack_await_time` and its `unused_verifier_lifetime`,
632                        // and `tombstone_retention_window` must clear the former — so exposing
633                        // either while `reconstructor` above stays pinned to upstream's defaults
634                        // would only be a way to break a pairing `validate` then rejects. Three are
635                        // unreachable below a batch size of two, which is not hoprd's default. The
636                        // last is a latency-versus-exposure dial with no second half to disagree
637                        // with, left alone because upstream's number is sized against the deployed
638                        // profile and hoprd has nothing to add to it.
639                        max_failed_cycles: supervision_defaults.max_failed_cycles,
640                        max_recovery_idle: supervision_defaults.max_recovery_idle,
641                        max_off_front_share_fraction: supervision_defaults
642                            .max_off_front_share_fraction,
643                        min_share_order_sample: supervision_defaults.min_share_order_sample,
644                        max_predeposit_packets: supervision_defaults.max_predeposit_packets,
645                        tombstone_retention_window: supervision_defaults.tombstone_retention_window,
646                    },
647                },
648                path_planner: Default::default(),
649                counter_flush_interval: HoprProtocolConfig::default().counter_flush_interval,
650                // Same standing as `counter_flush_interval` above: an interval at which telemetry
651                // is folded into the network graph, not a knob hoprd exposes. Taken from upstream's
652                // default rather than restated, so the number lives in one place.
653                surb_flush_interval: HoprProtocolConfig::default().surb_flush_interval,
654                mixer: value.network.mixer,
655                stream: Default::default(),
656            },
657            ..Default::default()
658        }
659    }

668/// The main configuration object of the entire node.
669///
670/// The configuration is composed of individual configurations of corresponding
671/// component configuration objects.
672///
673/// A default configuration YAML can be generated via `hoprd-cfg --default`.
674#[derive(
675    Debug, Serialize, Deserialize, Validate, Clone, PartialEq, smart_default::SmartDefault,
676)]
677#[serde(deny_unknown_fields)]
678pub struct HoprdConfig {
679    /// Configuration related to hopr-lib functionality
680    #[validate(nested)]
681    #[serde(default)]
682    pub hopr: UserHoprLibConfig,
683    /// Configuration regarding the identity of the node
684    #[validate(nested)]
685    #[serde(default)]
686    pub identity: Identity,
687    /// Configuration of the underlying database engine
688    #[validate(nested)]
689    #[serde(default)]
690    pub db: Db,
691    /// Configuration relevant for the API of the node
692    #[validate(nested)]
693    #[serde(default)]
694    pub api: Api,
695    /// Configuration of the Session entry/exit node IP protocol forwarding.
696    #[validate(nested)]
697    #[serde(default)]
698    pub session_ip_forwarding: SessionIpForwardingConfig,
699    /// Blokli provider URL to connect to.
700    #[validate(url)]
701    #[default(default_blokli_url())]
702    pub blokli_url: String,
703    /// Optional override for the DNS resolution of the Blokli provider
704    #[serde(default)]
705    pub blokli_dns_override: Option<(IpAddr, Option<u16>)>,
706    /// Configuration of underlying node behavior in the form strategies
707    ///
708    /// Strategies represent automatically executable behavior performed by
709    /// the node given pre-configured triggers.
710    #[validate(nested)]
711    #[serde(default = "crate::strategy::hopr_default_strategies")]
712    #[default(crate::strategy::hopr_default_strategies())]
713    pub strategy: crate::strategy::MultiStrategyConfig,
714}

716impl HoprdConfig {
717    pub fn as_redacted(&self) -> Self {
718        let mut ret = self.clone();
719        // redacting sensitive information
720        match ret.api.auth {
721            Auth::None => {}
722            Auth::Token(_) => ret.api.auth = Auth::Token("<REDACTED>".to_owned()),
723        }
724
725        if ret.identity.private_key.is_some() {
726            ret.identity.private_key = Some("<REDACTED>".to_owned());
727        }
728
729        "<REDACTED>".clone_into(&mut ret.identity.password);
730
731        ret
732    }
733
734    pub fn as_redacted_string(&self) -> anyhow::Result<String> {
735        let redacted_cfg = self.as_redacted();
736        Ok(serde_json::to_string(&redacted_cfg)?)
737    }
